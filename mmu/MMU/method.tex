\section{Method}

\paragraph{Matryoshka Representation Learning.}
MRL turns a single-scale objective into a multi-scale one. For nesting
dimensions $\mathcal{M} = \{m_1 < \dots < m_K\}$ and a representation
$f_\theta : \mathcal{X} \to \mathbb{R}^{d}$, it optimises the same loss at every
$m \in \mathcal{M}$ through a per-granularity linear classifier
$W^{(m)} \in \mathbb{R}^{L \times m}$, aggregating with importance scales
$c_m \ge 0$:
\begin{equation}
  \min_{\{W^{(m)}\}_{m \in \mathcal{M}},\, \theta} \;
  \frac{1}{N} \sum_{i \in [N]} \sum_{m \in \mathcal{M}}
  c_m \cdot \mathcal{L}\Bigl( W^{(m)} \bigl[f_\theta(x_i)\bigr]_{1:m} \,;\, y_i \Bigr).
  \label{eq:mrl}
\end{equation}
We follow the paper in setting $c_m = 1$ for all $m$. MMU applies
\eqref{eq:mrl} to SCRUB's objective instead of to cross-entropy alone.

\paragraph{Setup.}
Let $(W, c) \in \mathbb{R}^{L \times d} \times \mathbb{R}^{L}$ be the source
model's classifier, $\phi^{o}$ the frozen source (teacher) parameters,
$\mathcal{D}_f$ the forget set and $\mathcal{D}_r$ the retain set. The nesting
set is $\mathcal{M} = \{d/2^{4}, \dots, d/2, d\}$ restricted to $m \ge 8$, with
$m_K = d$.

\paragraph{Nested readouts.}
The logits at width $m$ are
\begin{equation}
  z^{(m)}(x) \;=\;
  \begin{cases}
    W^{(m)} \bigl[f_\theta(x)\bigr]_{1:m} + c^{(m)},
      & \textbf{MRL (default)}: \text{a separate classifier per } m,\\[4pt]
    W_{:,1:m} \bigl[f_\theta(x)\bigr]_{1:m} + c,
      & \textbf{MRL-E}: \text{one matrix sliced by column},
  \end{cases}
  \label{eq:readout}
\end{equation}
with $z^{(d)} = W f_\theta(x) + c$ in both cases, i.e.\ the model's ordinary
output. The auxiliary classifiers of the MRL form are warm-started at
$W^{(m)} \!=\! W_{:,1:m}$, $c^{(m)} \!=\! c$ -- so at step $0$ they are the
model's own prefix readouts rather than random ones -- are trained jointly with
$\theta$ by \eqref{eq:maxstep}--\eqref{eq:minstep}, and are discarded when
unlearning ends; the returned model is an ordinary classifier read at $m = d$.
The MRL-E form adds no parameters at all and is reported as an ablation.

\paragraph{Nested distillation.}
The teacher is the source model, which has no per-width classifiers, so it is
always read by column slicing: $\tilde z^{(m)}(x) = W_{:,1:m}
[f_{\theta^{o}}(x)]_{1:m} + c$. With $\sigma_T(\cdot) = \mathrm{softmax}(\cdot/T)$,
\begin{equation}
  \mathcal{K}^{(m)}(\mathcal{B})
  = \frac{T^{2}}{|\mathcal{B}|} \sum_{x \in \mathcal{B}}
    \mathrm{KL}\Bigl( \sigma_T\bigl(\tilde z^{(m)}(x)\bigr) \,\Big\|\,
                      \sigma_T\bigl(z^{(m)}(x)\bigr) \Bigr),
  \qquad
  \mathcal{C}^{(m)}(\mathcal{B})
  = \frac{1}{|\mathcal{B}|} \sum_{(x,y) \in \mathcal{B}}
    \ell_{\mathrm{CE}}\bigl(z^{(m)}(x), y\bigr).
  \label{eq:terms}
\end{equation}

\paragraph{MMU objective.}
SCRUB carries two losses, so both are nested by \eqref{eq:mrl}: a \emph{max-step}
that drives every nesting dimension away from the teacher on $\mathcal{D}_f$ and
a \emph{min-step} that restores every nesting dimension on $\mathcal{D}_r$,
\begin{align}
  \mathcal{L}_{\mathrm{max}}(\mathcal{B}_f)
    &= -\sum_{m \in \mathcal{M}} c_m \, \mathcal{K}^{(m)}(\mathcal{B}_f),
    \label{eq:maxstep}\\[2pt]
  \mathcal{L}_{\mathrm{min}}(\mathcal{B}_r)
    &= \phantom{-}\sum_{m \in \mathcal{M}} c_m
       \Bigl[\, \gamma \, \mathcal{C}^{(m)}(\mathcal{B}_r)
              + \alpha \, \mathcal{K}^{(m)}(\mathcal{B}_r) \Bigr].
    \label{eq:minstep}
\end{align}
Setting $\mathcal{M} = \{d\}$ collapses \eqref{eq:maxstep}--\eqref{eq:minstep}
to SCRUB's max- and min-steps exactly, so MMU is a strict generalisation of
SCRUB along the nesting axis.

\paragraph{Reweighted variant (ablation).}
With $c_m = 1$ the objective carries $K$ times SCRUB's gradient mass. The
\emph{normalised} variant redistributes a fixed mass instead, ranking the
max-step innermost-first and the min-step by relative width:
\begin{equation}
  c^{\mathrm{f}}_{m_k} = \frac{k^{-\gamma_f}}{\sum_{j=1}^{K} j^{-\gamma_f}},
  \qquad
  c^{\mathrm{r}}_{m} = \frac{(d/m)^{\beta_r}}{\sum_{m' \in \mathcal{M}} (d/m')^{\beta_r}},
  \qquad
  \textstyle\sum_k c^{\mathrm{f}}_{m_k} = \sum_m c^{\mathrm{r}}_{m} = 1,
  \label{eq:weights}
\end{equation}
so that $\gamma_f$ and $\beta_r$ change only how the stability/forgetting budget
is distributed over widths, not how large it is.

\paragraph{Per-width diagnostic.}
Residual knowledge is reported at every nesting dimension rather than only at
$m = d$:
\begin{equation}
  A^{(m)}(\mathcal{D}) = \frac{1}{|\mathcal{D}|}
  \sum_{(x,y) \in \mathcal{D}} \mathbf{1}\Bigl[\arg\max_{\ell} z^{(m)}_{\ell}(x) = y\Bigr],
  \qquad m \in \mathcal{M},
\end{equation}
measured on $\mathcal{D}_f$ and $\mathcal{D}_r$ before and after unlearning.

\begin{algorithm}[t]
\caption{MMU: SCRUB under MRL Eq.~\eqref{eq:mrl}}
\label{alg:mmu}
\begin{algorithmic}[1]
  \Require source $\phi^{o} = (\theta^{o}, W, c)$, sets $\mathcal{D}_f,
    \mathcal{D}_r$, widths $\mathcal{M}$, epochs $E$, max-step epochs
    $E_{\max} \le E$, rate $\eta$, temperature $T$, weights $\alpha, \gamma$,
    scales $\{c_m\}$
  \State $\theta \gets \theta^{o}$; freeze teacher $\phi^{o}$
  \State \textbf{MRL:} $W^{(m)} \gets W_{:,1:m},\; c^{(m)} \gets c$ for every
         $m \in \mathcal{M} \setminus \{d\}$ \Comment{warm start; MRL-E: skip}
  \For{$e = 1, \dots, E$}
    \If{$e \le E_{\max}$}
      \For{minibatch $\mathcal{B}_f \subset \mathcal{D}_f$}
        \State descend $\nabla \mathcal{L}_{\mathrm{max}}(\mathcal{B}_f)$ in
               $\theta$ and $\{W^{(m)}\}$ \Comment{Eq.~\eqref{eq:maxstep}}
      \EndFor
    \EndIf
    \For{minibatch $\mathcal{B}_r \subset \mathcal{D}_r$}
      \State descend $\nabla \mathcal{L}_{\mathrm{min}}(\mathcal{B}_r)$ in
             $\theta$ and $\{W^{(m)}\}$ \Comment{Eq.~\eqref{eq:minstep}}
    \EndFor
  \EndFor
  \State discard $\{W^{(m)}\}_{m < d}$
  \State \Return $(\theta, W, c)$
\end{algorithmic}
\end{algorithm}
